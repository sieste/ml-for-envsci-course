\begin{frame}{Supervised machine learning: General setup}
\begin{itemize}
\item We have an observable \textbf{target outcome} we wish to predict. 
\item The prediction can be based on a set of \textbf{input features}. 
\item For example
\begin{itemize}
\item predict future temperature based on current temperature, pressure, wind 
\item predict occurrence of wildfire based on measures of heat, humidity, wind 
\item predict crop yield based on agricultural practice, soil nutrients, pests
\end{itemize}
\item The goal of supervised ML is to \textbf{build a model} that predicts outcomes based on available features.
\item A "good" ML model is one that produces \textbf{accurate predictions}.
\end{itemize}
\end{frame}


\begin{frame}{Regression and classification}
\begin{itemize}
\item The type of target variable determines the type of ML problem.
\item \textbf{Regression:} Predicting quantitative outcomes. 
\begin{itemize}
\item For example: temperature in degrees, crop yield in tonnes, number of sharks, rain amount in mm
\end{itemize}
\item \textbf{Classification:} Predicting categorical outcomes. 
\begin{itemize}
\item For example: wildfire yes/no, temperature freezing/nonfreezing, precipitation type rain/snow/hail, air quality low/medium/high
\end{itemize}
\item Regression vs classification determines various details of the ML model and analysis.
\end{itemize}
\end{frame}

\subsection{Model training}

\begin{frame}{Function approximation theory}
\begin{itemize}
\item A ML model is a \textbf{prediction rule} that translates one or more features \(x_1, x_2, \dots\) into an estimate \(\hat{y}\) of the target \(y\).
\item The prediction rule is calculated by a \textbf{mathematical function} \[\hat{y} = f(x_1, x_2, \dots)\]
\item The function \(f(\dots)\) usually includes \textbf{model parameters} \(\theta_1, \theta_2, \dots\) that control how inputs are translated to outputs.
\item Defining the feature vector \(\fat{x} = (x_1, x_2, \dots)\) and parameter vector \(\fat{\theta}=(\theta_1, \theta_2, \dots)\) a trainable ML model is often compactly written as \[\hat{y} = f(\fat{x}; \fat\theta)\]
\end{itemize}
\end{frame}


\begin{frame}{Training data}
\begin{itemize}
\item Supervised ML works by "learning from examples".
\item \textbf{Training data} \(S\) is a set of input-output examples \[S = \left\{(\fat{x}_i, y_i)\right\}_{i=1}^n\] where
\begin{itemize}
\item \(\fat{x}_i = (x_{i,1}, x_{i,2}, \dots, x_{i,k})\) is the \textbf{feature vector} (containing \(k\) features) in the \(i\)-th example
\item \(y_i\) is the \textbf{target outcome} observed in the \(i\)-th example
\item \(n\) is the \textbf{sample size}
\end{itemize}
\item The data set \(S\) is used to train our ML model \(f(\fat{x};\fat\theta)\) that can predict the outcome \(y^*\) for arbitrary feature vector \(\fat{x}^*\).
\end{itemize}
\end{frame}


\begin{frame}{Loss function}
\begin{itemize}
\item A loss function \(L(\hat{y},y)\) is a function that quantifies \textbf{how different} the prediction \(\hat{y}\) is to the target outcome \(y\).
\item Loss functions are typically \textbf{negatively oriented}, where lower values indicate better predictions.
\item The loss function must be appropriate for the ML task at hand (regression vs classification) and can usually be set when intialising the ML model.
\item Otherwise the choice of loss function is \textbf{often ambiguous} and difficult to fully justify a particular choice.
\end{itemize}
\end{frame}


\begin{frame}{Loss function}
\begin{itemize}
\item \textbf{Regression} (where \(\hat{y}\) and \(y\) are real numbers):
\begin{itemize}
\item Squared-error loss \(L(\hat{y},y) = (\hat{y} - y)^2\)
\item Absolute-error loss \(L(\hat{y},y) = |\hat{y} - y|\)
\end{itemize}
\item \textbf{Binary classification} (where \(y\) is 0 or 1 and \(\hat{y}\) is the probability for outcome "1"):
\begin{itemize}
\item Cross-entropy loss \(L(\hat{y},y) = -y\log\hat{y} - (1-y)\log(1-\hat{y})\)
\item Quadratic loss \(L(\hat{y},y) = (\hat{y} - y)^2\)
\end{itemize}
\item \textbf{Multivariate classification} (where \(y = (0,\dots,1,\dots,0)\) is a binary vector indicating which of \(k\) classes occurs and \(\hat{y}\) is a vector of \(k\) class probabilities)
\begin{itemize}
\item Cross-entropy loss \(L(\hat{y},y) = - \sum_{i=1}^k y_i \log \hat{y}_i\)
\item Focal loss \(L(\hat{y},y) = - \sum_{i=1}^k y_i (1-\hat{y}_i)^\gamma \log \hat{y}_i\)
\end{itemize}
\end{itemize}
\end{frame}



\begin{frame}{Empirical loss minimisation}
\begin{itemize}
\item We can now \textbf{formalise} training of ML model \(f(\fat{x};\fat\theta)\) as
\begin{itemize}
\item trying to find model parameters \(\fat\theta\) such that
\item the model outputs \(\hat{y}_1, ..., \hat{y}_n\) calculated from features \(\fat{x}_1, \dots, \fat{x}_n\) 
\item achieve minimal average loss when compared to the targets \(y_1, \dots, y_n\) in the training data set
\item using an appropriate loss function \(L(\hat{y}, y)\).
\end{itemize}
\item \textbf{The Empirical Loss Minimisation paradigm for ML:}
\[\hat{\fat\theta} = \underset{\fat\theta}{\mathrm{argmin}}\frac1n\sum_{i=1}^n L\left[f(\fat{x}_i; \fat\theta), y_i\right]\]
\item The trained model with parameters \(\hat{\fat\theta}\) can then be used to make predictions \(\hat{y}^*\) based on new feature vectors \(\fat{x}^*\)
\[\hat{y}^* = f(\fat{x}^*; \hat{\fat\theta})\]
\end{itemize}
\end{frame}

\begin{frame}{In summary ...}
To train a ML model by empirical loss minimisation we need:
\begin{itemize}
\item a training data set \(S\) containing \(n\) examples of inputs and corresponding outcomes
\item a model function \(f(\fat{x};\fat\theta)\) with trainable parameters \(\fat\theta\) that translates inputs \(\fat{x}\) into an output \(\hat{y}\)
\item a loss function \(L(\hat{y},y)\) that quantifies how close model output \(\hat{y}\) is to the target outcome \(y\)
\item a search algorithm to find parameters that minimise the loss 
\end{itemize}
The combination of training data set, search algorithm, model function \(f(\fat{x};\fat\theta)\) and loss function \(L(\hat{y},y)\) determine the ML approach and \textbf{these details should always be reported when documenting your ML model development.}
\end{frame}

\begin{frame}{How good is my model?}
\begin{itemize}
\item The model outputs \(\hat{y}_1, \dots, \hat{y}_n\) must be compared to the targets \(y_1, \dots, y_n\) using a suitable error metric.
\item For example root mean squared error (RMSE) \[RMSE = \sqrt{\frac1n \sum_{i=1}^n (\hat{y}_i - y_i)^2}\]
\item Our model has \(RMSE \approx 0.9\) -- Is that "good"?
\item Getting \(RMSE > 0\) means our model is not perfect, but how bad is it?
\item To judge whether the model is useful we should compare it to a suitable benchmark. 
\end{itemize}
\end{frame}


\begin{frame}{Benchmarking}
\begin{itemize}
\item A benchmark model (or reference model) for performance evaluation is usually one or several of
\begin{itemize}
\item An alternative (competing) model for the same target.
\item Climatology (constant mean): \[\hat{y}^{(clim)}_i = \frac1n \sum_{i=1}^n y_i\]
\item Persistence (last available observation of the target); here: \[\hat{y}^{(pers)}_i = y(t_i - \Delta)\]
\item Any other simple estimate of the target that could be calculated with reasonable effort given the same inputs as our model.
\item In fact, linear regression itself is often a good reference to benchmark more complicated ML model.
\end{itemize}
\end{itemize}
\end{frame}

\begin{frame}{Skill scores}
\begin{itemize}
\item Given a suitable error metric, we calculate
\begin{itemize}
\item \(S\): the error of our model (averaged over a test data set)
\item \(S_{ref}\): the mean error of our chosen reference model 
\item \(S_{perf}\): the mean error or a hypothetical perfect model that outputs \(\hat{y}_i = y_i\) each time; (usually \(S_{perf} = 0\))
\end{itemize}
\item The skill score of our model relative to the benchmark is then defined as
\[Skill = \frac{S_{ref} - S}{S_{ref} - S_{perf}}\]
\item \(Skill \le 0\): Our model is not better than the benchmark.
\item \(Skill \in (0, 1)\): Our model improves over the benchmark. 
\item \(Skill = 1\): Our model is perfect.
\end{itemize}
\end{frame}

\begin{frame}{Example: k-nearest-neighbor classification}
\begin{itemize}
\item given an input \(\fat{x}^*\) find the \(k\) closest features \(\fat{x}^{(1)}, \dots, \fat{x}^{(k)}\) in the training data, and the corresponding outcomes \(y^{(1)}, \dots, y^{(k)}\)
\item "closeness" between the feature vectors \(\fat{x}^{(i)}\) and \(\fat{x}^{(j)}\) is defined in terms of the Euclidean distance \(\sqrt{\sum_m (x^{(i)}_m - x^{(j)}_m)^2}\)
\item predict \(\hat{y}^*\) as the majority vote over \(y^{(1)}, \dots, y^{(k)}\)
\item ties are resolved by picking the outcome class lowest value or alphabetical rank
\end{itemize}
\end{frame}

\begin{frame}{In-sample vs Out-of-sample error}
\begin{itemize}
\item Our goal is to train a model that generalises well to new data. 
\item Loss on training data (in-sample error) is often too optimistic because the model "has seen" the data.
\item Overfitting: A model achieves very low training loss by "regurtitating" the training data, including any noise and accidental patterns. Overfitted models generalise poorly to new data.
\item Loss on previously unseen data (out-of sample error) is what we are really interested in.
\item It is good practice to remove a fraction of data (e.g. 20\%) during training and use only for model testing.
\end{itemize}
\end{frame}

\begin{frame}{Decision tree terminology}
\begin{itemize}
\item \textbf{Split}: Partitioning data based on a feature and threshold 
\item \textbf{Node}: a point in the tree where a decision is made
\item \textbf{Root node}: first split in the tree
\item \textbf{Internal node}: A node that performs a split
\item \textbf{Leaf node}: Final node that outputs a prediction
\item \textbf{Depth}: Number of decision levels from Root node to leaves
\item \textbf{Impurity}: Measure how mixed the targets are within a split
\item \textbf{Information gain}: Reduction in impurity after a split
\end{itemize}
\end{frame}

\begin{frame}{Finding features and thresholds for splitting}
\begin{minipage}{.6\textwidth}
\begin{itemize}
\item Gini impurity and entropy are measures of impurity, disorder, indeterminism.
\item Binary data: \(\fat{y} = (y_1, \dots, y_n) = (0,0,1,0,1,...)\).
\item Summarise \(\bar{y} = \frac1n \sum_{i=1}^n y_i\).
\end{itemize}
\end{minipage}%
\begin{minipage}{.4\textwidth}
\includegraphics[width=\textwidth]{fig/impurity-sketch.png}
\end{minipage}%

\vfill

\begin{itemize}
\item Gini impurity: \(G(\fat{y}) = 1 - \bar{y}^2 - (1-\bar{y})^2\)
\item Entropy: \(H(\fat{y}) = \bar{y}\log \bar{y} + (1-\bar{y})\log (1-\bar{y})\)
\end{itemize}
\end{frame}


\begin{frame}{Maximising Information Gain}
\begin{itemize}
\item We apply threshold-splitting of the data wrt one feature.
\item This separates the data \(\fat{y}\) into \(\fat{y}_{left}\) and \(\fat{y}_{right}\) of sizes \(n_{left}\) and \(n_{right}\).
\item The impurity of the split data set is calculated by \[G(\fat{y}_{left}, \fat{y}_{right}) = \frac{n_{left}}{n} G(\fat{y}_{left}) + \frac{n_{right}}{n} G(\fat{y}_{right})\]
\item The impurity reduction (or "information gain" \(IG\)) caused by the split is given by \[IG = G(\fat{y}) - G(\fat{y}_{left}, \fat{y}_{right})\]
\item The feature/threshold combination that yields the highest information gain is selected.
\end{itemize}
\end{frame}

\begin{frame}{Feature importance}
\begin{itemize}
\item To interpret a trained ML model it is often interesting to ask "which feature contributes most to the model skill?"
\item This is a very difficult question when there are complex interactions between features.
\item Some options include
\begin{itemize}
\item Quick and dirty, for decision trees: \texttt{clf.feature\_importances\_} (mean impurity decrease)
\item More informative and generic: \texttt{sklearn.inspection.permutation\_importance} (performance drop when a feature is randomly shuffled)
\item Widely used, but complicated, measure for reporting: SHAP values (global importance measure)
\item Many more (further reading)
\end{itemize}
\end{itemize}
\end{frame}


\begin{frame}{Hyperparameters}
\begin{minipage}{.7\textwidth}
\begin{itemize}
\item A hyperparameter is a model parameter that is not inferred from the training data by the learning algorithm.
\item For decision trees, model parameters are the decision thresholds, and hyperparameters can be
\begin{itemize}
\item minimum/maximum tree depth
\item maximum number of leaf nodes
\item minimum number of samples per split
\item impurity measure

\begin{frame}{Hyperparameter optimisation}
\begin{itemize}
\item Cross-validation: 
\begin{itemize}
\item randomly split the training data into \(n_{CV}\) subsets (usually 5 or 10)
\item use \(n_{CV} - 1\) subsets to train the model 
\item evaluate the model on the left-out subset
\item repeat for each of the \(n_{CV}\) folds
\item average the \(n_{CV}\) out-of-sample losses into a single cross-validation loss
\end{itemize}
\item Grid search:
\begin{itemize}
\item Enumerate all \(n_{comb}\) combinations of hyperparameters
\item Calculate cross validation loss for each combination
\item Choose the hyperparameter setting with lowest cv loss
\item Fit the model with the optimised hyperparameters
\end{itemize}
\item The computational cost of exhaustiv grid search with cross validation is \(n_{CV} \times n_{comb}\) larger than training a single model.
\end{itemize}
\end{frame}

\begin{frame}{Boosting}
\begin{itemize}
\item Ensemble methods (such as RF) train a number of \textbf{strong} classifiers \textbf{independently}.
\item \textbf{Boosting methods} trains a chain of \textbf{weak} classifiers \textbf{sequentially}.
\item A "weak learner" is simple, fast to train, and has low individual accuracy.
\item Core algorithm:
\begin{itemize}
\item train a simple model
\item calculate the model's errors over the training data
\item train the next model to take the same inputs but predict the previous model's errors
\item repeat, then sum the outputs of all models (often with weights)
\end{itemize}
\item Throughout the boosting sequence the model is improved by gradually refining the prediction.
\end{itemize}
\end{frame}


\begin{frame}{Gradient Boosting}
Technical details of gradient boosting:
\begin{itemize}
\item \textbf{Gradients} of the loss function \(g_i = \frac{\partial L(\hat{y}_i,y_i)}{\partial \hat{y}_i}\) approximate prediction errors/residuals.
\item Changing \(\hat{y}_i\) proportional to \(-g_i\) will decrease the loss.
\item In \textbf{gradient boosting}, each tree predicts the negative gradient \(\frac{\partial L(\hat{y},y)}{\partial \hat{y}}\) of the loss of the previous tree.
\item If \(f_{i+1}(\fat{x};\fat\theta_{i+1})\) is the output of the \(i+1\)-st learner, the \(i+1\)st model prediction is \[\hat{y}_{i+1} = \hat{y}_i + \eta f_{i+1}(\fat{x};\fat\theta_{i+1})\] where \(\eta\) is the \textbf{learning rate}.
\end{itemize}
\end{frame}

\subsection{Multilayer Perceptron}


\begin{frame}{Artificial neuron (Perceptron)}
\begin{itemize}
\item An artificial neuron (perceptron) is a mathematical function 
\[\hat{y} = f(\fat{x}; \fat{w}, b) = \varphi\left( w_1 \cdot x_1 + \dots + w_k \cdot x_k + b\right)\]
\item where
\begin{itemize}
\item \(\fat{x} = (x_1, \dots, x_k)\) is the \(k\)-dimensional input vector
\item \(y\) is the 1-dimensional output of the neuron
\item \(\fat{w} = (w_1, \dots, w_k)\) is the \(k\)-dimensional vector of weights 
\item \(b\) is the bias
\item \(\varphi(\cdot)\) is the activation function
\end{itemize}
\item A neuron takes inputs \(\fat{x}=(x_1, \dots, x_k)\), calculates a \textbf{linear combination}, and applies a \textbf{non-linear activation function}.
\end{itemize}
\end{frame}

\begin{frame}{Activation functions}
\begin{itemize}
\item Typical activation functions:
\begin{itemize}
\item ReLU (Rectifier): \(\varphi(x) = max(0,x)\)
\item Sigmoid: \(\varphi(x) = 1 / (1 + e^{-x})\)
\item Swish: \(\varphi(x) = x / (1 + e^{-x})\)
\item Tanh: \(\varphi(x) = (e^x - e^{-x}) / (e^x + e^{-x})\)
\end{itemize}
\end{itemize}
\begin{center}
\includegraphics[width=.5\textwidth]{fig/nn-activations.png}
\end{center}
\end{frame}


\begin{frame}{Single-layer perceptron}
\begin{itemize}
\item In a SLP, \textbf{several perceptrons} (each with different weights and biases) calculating several outputs in parallel
\item produce a \textbf{vector of outputs} \(y_1,\dots,y_N\) where
\[y_i = \varphi\left(w_{i,1} x_1 + \dots + w_{i,k} x_k + b_i\right) = \varphi\left(\fat{w_i} \cdot \fat{x} + b_i\right)\]
\end{itemize}
\begin{center}
\includegraphics[width=.5\textwidth]{fig/nn-single-layer-perceptron.png}
\end{center}
\end{frame}

\begin{frame}{Multilayer perceptron}
\begin{minipage}{.6\textwidth}
\begin{itemize}
\item In an MLP, \textbf{perceptrons are arranged in multiple layers}
\item Each layer computes a \textbf{vector of activations} that serves as input to the next layer
\item For layer $\ell$, the vector of outputs \(\fat{y}^{(\ell)}=(y^{(\ell)}_1, \dots y^{(\ell)}_{n_\ell})\) is calculated from the previous one by
\[
y^{(\ell)}_i = \varphi\left(\fat{w}^{(\ell)}_i \cdot \fat{y}^{(\ell-1)} + b^{(\ell)}_i\right)
\]
\end{itemize}

\begin{frame}{Terminology}
\begin{itemize}

\item \textbf{Input layer}: First layer that accepts the input features

\item \textbf{Hidden layer(s)}: layer(s) between input and output layers
\item \textbf{Parameters}: all trainable weights and biases of the network
\item \textbf{Forward pass}: computation of outputs from inputs by the network
\end{itemize}

\subsection{Convolutional Neural Networks}


\begin{frame}{Convolution}
\begin{minipage}{.6\textwidth}
\begin{itemize}
\item Perceptrons are \textbf{densely connected} layers because every node in a layer receives outputs from all nodes in the previous layer.
\item When inputs and outputs are high-dimensional, the number of trainable parameters in a dense layer becomes very large. 
\item Convolution is a sparsely connected layer with where outputs are only connected to \textbf{spatially close inputs} and weights are \textbf{shared across the domain}.
\end{itemize}
\end{minipage}%
\begin{minipage}{.4\textwidth}
\begin{center}
\includegraphics[width=.9\textwidth]{fig/dense-vs-local.png}
\end{center}
\end{minipage}%
\end{frame}



\begin{frame}{Convolution}
\begin{itemize}
\item The local, shared combination weights are stored in a \textbf{convolution kernel}
\item Usually kernels are \(3\times 3\) or \(5 \times 5\), we're using the less common \(2\times 2\) for illustration here.
\item Boundary conditions need to be considered later.
\end{itemize}
\begin{center}
\includegraphics[width=.5\textwidth]{fig/cnn-illustration-1.png}
\end{center}
\end{frame}

\begin{frame}{Convolution}
\begin{itemize}
\item The input data usually have \textbf{multiple channels} (variables, color channels, pressure levels)
\item The convolution kernel must have the same number of channels as the input data.
\end{itemize}
\vfill
\begin{center}
\includegraphics[width=.6\textwidth]{fig/cnn-illustration-2.png}
\end{center}
\end{frame}

\begin{frame}{Convolution}
\begin{itemize}
\item By defining multiple convolution kernels, we produce \textbf{multiple output channels} from the same input channels.
\end{itemize}
\vfill
\begin{center}
\includegraphics[width=.6\textwidth]{fig/cnn-illustration-3.png}
\end{center}
\begin{itemize}
\item This is the \textbf{typical situation for 2d convolution} in a convolutional neural network.
\item The \textbf{trainable parameters} in a 2d convolution are stored in a 4d data structure with dimensions (width, height, input channels, output channels).
\end{itemize}
\end{frame}



\begin{frame}{Convolution}

\textbf{Convolution mathematically}:

\begin{itemize}
\item inputs \(x[i,j,c]\) indexed by space \((i,j)\) and input channel \(c=1,\dots,C\)
\item outputs \(y[i,j,d]\) indexed by space \((i,j)\) and output channel \(d=1,\dots,D\)
\item kernel \(k[m,n,c,d]\) of width and height \(m = -w,\dots,+w\), \(C\) input channels, \(D\) output channels
\end{itemize}
\[y[i,j,d] = \sum_{m = -w}^{w} \sum_{n = -w}^{w} \sum_{c=1}^C k[m,n,c,d] \cdot x[i+m,j+n,c]\]
\end{frame}


\begin{frame}{Convolution}
\begin{itemize}
\item 2D-convolutions can be chained into layers of a \textbf{Deep Convolutional Neural Network} (Deep CNN).
\end{itemize}
\vfill
\begin{center}
\includegraphics[width=.9\textwidth]{fig/cnn-illustration-4.png}
\end{center}
\end{frame}


\begin{frame}{Typical setup}
\begin{itemize}
\item A common situation in ML with spatial data:
\begin{itemize}
\item several input variables, each a 2d spatial field (input channels)
\item the target is a single spatial field of the same size (output channel)
\end{itemize}
\item A typical CNN for such data has:
\begin{itemize}
\item 2-6 hidden convolution layers
\item kernel size 3 or 5
\item 16-256 channels per hidden layer
\item 1 channel at output layer (to match the target)
\end{itemize}
\item This will produce models with \(10^4\) - \(10^6\) parameters, suitable for training locally on moderately large training data sets
\item These hyperparameters can be explored and optimised as part of the ML model development.
\end{itemize}
\end{frame}

\begin{frame}{Training Neural Networks}
\textbf{Gradient descent algorithm:}
\begin{enumerate}
\item Initialise model weights randomly.
\item Repeat until convergence:
\begin{itemize}
    \item Calculate \textbf{gradient} of the loss function with respect to model weights:
\begin{equation*}
\frac{\partial L}{\partial \mathbf{w}}(\mathbf{w}) = \begin{pmatrix}\frac{\partial L(\mathbf{w})}{\partial w_1}\\ \vdots \\ \frac{\partial L(\mathbf{w})}{\partial w_k}\end{pmatrix}
\end{equation*}
\item \textbf{Adjust weights} in the opposite direction of the gradient:
\begin{equation*}
\mathbf{w}^{t+1} \leftarrow \mathbf{w}^t - \gamma \frac{\partial L}{\partial \mathbf{w}}(\mathbf{w}^t)
\end{equation*}
where $\gamma$ is called the \textbf{learning rate}
\end{itemize}
\item Return weights.
\end{enumerate}
\end{frame}


\begin{frame}[fragile]{Stochastic Gradient Descent (SGD)}
\begin{itemize}
\item In Gradient Descent, the loss and its gradient are calculated on the whole training data set.
\item In SGD, loss and gradient are calculated on random subsets (\textbf{batches}) of the whole training data set.
\item Random subsetting reduces computational cost of gradient calculation and improves convergence of the optimiser.
\item The time taken to loop over the entire training data set once is called an \textbf{epoch}.
\item A variant of SGD called the \textbf{Adam optimiser} is commonly used.
\end{itemize}
\vfill
\begin{lstlisting}[language=Python]
# optimiser, loss are specified during initialisation
model.compile(optimizer='adam', loss='mse')

# epochs and batch size are specified during training
model.fit(X, y, epochs=100, batch_size=64)
\end{lstlisting}
\end{frame}




\begin{frame}{Backpropagation}
\begin{itemize}
\item "Backprop" is an efficient algorithm to calculate the gradient of the loss with respect model weights
\item Propagates loss backward from output to input layers using the chain rule of differentiation
\item Efficiency by reusing intermediate computations
\item More efficient than, e.g., finite differencing which requires separate loss evaluations for each weight
\item Backpropagation exploits automatic differentiation in modern ML libraries such as PyTorch and TensorFlow
\end{itemize}
\end{frame}

\begin{frame}[fragile]{Regularisation}
Weight-level methods to avoid overfitting:
\begin{itemize}
\item \textbf{Weight regularisation}: Add a term to the loss function that is proportional to the magnitude of the weights. Encouraging the optimiser to keep the weights small can avoid overfitting.
\item \textbf{Dropout}: During training a random subset of layer outputs are set to zero. The network learns "stable" representations of the data, and generalises better.
\end{itemize}

\begin{lstlisting}[language=Python]
from tensorflow.keras.layers import Dropout
from tensorflow.keras import regularizers

model = Sequential([
    Input(input_shape=(X_train.shape[1],)),
    Dense(64, activation="relu", kernel_regularizer=regularizers.l2(1e-4)),
    Dropout(0.3),
    Dense(32, activation="relu", kernel_regularizer=regularizers.l2(1e-4)),
    Dropout(0.3),
    Dense(1) ])
\end{lstlisting}
\end{frame}

\begin{frame}{Training a multi-class classifier}
\textbf{Softmax output}
\begin{itemize}
\item We want our model to output a vector of probabilities that sum to one.
\item This is achieved by the \textbf{softmax activation function} that takes arbitrary inputs \(x_1, \dots, x_k\) and outputs a vector with \(i\)-th element
\[softmax(x_1, \dots, x_k)_i = \frac{e^{x_i}}{\sum_{j=1}^k e^{x_j}}\]
\end{itemize}
\end{frame}


\begin{frame}{Training a multi-class classifier}
\textbf{One-hot-encoding}
\begin{itemize}
\item To train a multi-class classifier, we one-hot-encode our target labels.
\item If a target takes \(1\) out of \(k\) possible classes, the one-hot-encoding is a vector of length \(k\), where the element in the target class position is \(1\) and all other elements are \(0\)
\item For example, when there are \(5\) possible classes, a target with class label \(4\) is one-hot-encoded by the vector \[\fat{y} = (0, 0, 0, 1, 0)\]
\end{itemize}
\end{frame}

\begin{frame}{Training a multi-class classifier}
\textbf{Categorical cross entropy loss}
\begin{itemize}
\item Cross entropy is a loss function to train classifiers that output a vector of probabilities \(\hat{\fat{y}} = (\hat{y}_1, \dots, \hat{y}_k)\) of length \(k\) and the corresponding target \(\fat{y} = (y_1, \dots, y_k)\) is one-hot-encoded.
\item Cross entropy loss is given by \[L(\hat{\fat{y}}, \fat{y}) = -\sum_{i=1}^k y_i \log(\hat{y}_i)\]
\item In words: The negative logarithm of the probability assigned to the correct class.
\end{itemize}
\end{frame}

\begin{frame}[fragile]{Example: Animal image classification}
\textbf{Define model, optimiser, loss function}
\begin{lstlisting}[language=Python]
# define encoder architecture
model = Sequential([
    Input(shape=X_train.shape[1:]),
    Conv2D(64, (3, 3), activation="relu"),
    MaxPooling2D((2, 2)),
    Conv2D(32, (3, 3), activation="relu"),
    MaxPooling2D((2, 2)),
    Conv2D(16, (3, 3), activation="relu"),
    MaxPooling2D((2, 2)),
    Flatten(),
    Dense(128, activation="relu"),
    Dense(y.shape[1], activation="softmax")
])

model.compile(
    optimizer="adam",
    loss="categorical_crossentropy",
    metrics=["accuracy"]
)
